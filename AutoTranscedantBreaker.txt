-- Variables

-- Position data

    {name = "Position 1", vector = Vector3.new(23.192707061767578, -0.5, -27.228668212890625)},
    {name = "Position 2", vector = Vector3.new(0, 0, 0)}, -- Add more positions as needed

-- Get the remote events
local player = game:GetService("Players").LocalPlayer
local backpack = player:WaitForChild("Backpack")

local roarEvent = transcendentBreaker:WaitForChild("RoarEvent")
local judgmentEvent = transcendentBreaker:WaitForChild("JudgmentEvent")
local m1Event = transcendentBreaker:WaitForChild("M1Event")

-- Create dropdown items
local function createDropdownItems()
    for i, pos in ipairs(positions) do
        local item = Instance.new("TextButton")
        item.Name = "Item_" .. i
        item.Size = UDim2.new(1, 0, 0, 25)
        item.Position = UDim2.new(0, 0, 0, (i-1) * 25)
        item.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
        item.BackgroundTransparency = 0.3
        item.Text = pos.name
        item.TextColor3 = Color3.fromRGB(200, 200, 200)
        item.Font = Enum.Font.Gotham
        item.TextSize = 12
        item.ZIndex = 11
        item.Parent = dropdownList
        
        -- Hover effect
        item.MouseEnter:Connect(function()
            item.BackgroundColor3 = Color3.fromRGB(80, 80, 80)
        end)
        
        item.MouseLeave:Connect(function()
            item.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
        end)
        
        -- Selection handler
        item.MouseButton1Click:Connect(function()
            selectedPosition = i
            dropdownButton.Text = positions[i].name
            toggleDropdown()
        end)
    end
    
    dropdownList.CanvasSize = UDim2.new(0, 0, 0, #positions * 25)
end
createDropdownItems()

-- Toggle dropdown function
local function toggleDropdown()
    dropdownOpen = not dropdownOpen
    dropdownList.Visible = dropdownOpen
    arrowLabel.Text = dropdownOpen and "▲" or "▼"
end

-- Close dropdown when clicking outside
local userInputService = game:GetService("UserInputService")
userInputService.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        if dropdownOpen then
            local mousePos = userInputService:GetMouseLocation()
            local absolutePos = dropdownList.AbsolutePosition
            local absoluteSize = dropdownList.AbsoluteSize
            
            if mousePos.X < absolutePos.X or mousePos.X > absolutePos.X + absoluteSize.X or
               mousePos.Y < absolutePos.Y or mousePos.Y > absolutePos.Y + absoluteSize.Y then
                toggleDropdown()
            end
        end
    end
end)

-- Dropdown button click
dropdownButton.MouseButton1Click:Connect(toggleDropdown)

-- Function to fire all remotes
local function fireAllRemotes()
    -- Fire RoarEvent (no args)
    pcall(function()
        roarEvent:FireServer()
    end)
    
    -- Fire JudgmentEvent with position
    pcall(function()
        judgmentEvent:FireServer(positions[selectedPosition].vector)
    end)
    
    -- Fire M1Event (no args)
    pcall(function()
        m1Event:FireServer()
    end)
end

-- Spam function (0.2 second interval)
local function spamLoop()
    local spamCount = 0
    while isSpamming do
        fireAllRemotes()
        spamCount = spamCount + 1
        
        -- Update status every 10 iterations
        if spamCount % 10 == 0 then
            statusLabel.Text = string.format("Spamming: %d combos (%s)", spamCount, positions[selectedPosition].name)
        end
        
        -- Wait 0.2 seconds
        wait(0.2)
    end
end

-- Toggle function
local function toggleSpam()
    isSpamming = not isSpamming
    
    if isSpamming then
        -- Update UI for ON state
        toggleButton.Text = "🔴 STOP SPAMMING"
        toggleButton.BackgroundColor3 = Color3.fromRGB(200, 60, 60)
        statusLabel.Text = string.format("Spamming: 0 combos (%s)", positions[selectedPosition].name)
        statusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        speedLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        
        -- Start spamming
        spawn(spamLoop)
    else
        -- Update UI for OFF state
        toggleButton.Text = "START SPAMMING"
        toggleButton.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
        statusLabel.Text = "Status: IDLE"
        statusLabel.TextColor3 = Color3.fromRGB(200, 200, 200)
        speedLabel.TextColor3 = Color3.fromRGB(255, 200, 100)
    end
end

-- Connect button click
toggleButton.MouseButton1Click:Connect(toggleSpam)

-- Make GUI draggable
local dragging = false
local dragInput
local dragStart
local startPos

frame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = frame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

frame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)

userInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

-- Cleanup function
local function onDestroy()
    isSpamming = false
end

-- Handle when GUI is removed
screenGui.AncestryChanged:Connect(function()
    if screenGui.Parent == nil then
        onDestroy()
    end
end)
